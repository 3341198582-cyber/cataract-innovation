class SADDistillationDataset(Dataset):
    def __init__(self, transform=None):
        self.transform = transform
        self.data_infos = []
        class_map = {'normal': 0, 'cataract': 1}

        splits = ['train', 'test']
        for split in splits:
            for cls_name, label in class_map.items():
                roi_dir = os.path.join(roi_src, split, f"{cls_name}_ROI")
                mask_dir = os.path.join(mask_root, split, cls_name)
                npy_dir = os.path.join(msa_npy_dir, split, cls_name)
                
                if not os.path.exists(roi_dir): continue
                
                for f in os.listdir(roi_dir):
                    if f.lower().endswith(('.png', '.jpg', '.jpeg')):
                        file_name, file_ext = os.path.splitext(f)
                        
                        roi_path = os.path.join(roi_dir, f)
                        mask_path = os.path.join(mask_dir, f"{file_name}_SAMmask{file_ext}")
                        npy_path = os.path.join(npy_dir, f"{file_name}_msa.npy")
                        
                        if os.path.exists(mask_path) and os.path.exists(npy_path):
                            self.data_infos.append({
                                'roi_path': roi_path,
                                'mask_path': mask_path,
                                'npy_path': npy_path,
                                'label': label
                            })
        print(f"多模态数据对齐完成：成功配对 {len(self.data_infos)} 个样本")

    def __len__(self): return len(self.data_infos)

    def __getitem__(self, idx):
        item = self.data_infos[idx]
        
        img = cv2.imread(item['roi_path'])
        mask = cv2.imread(item['mask_path'], cv2.IMREAD_GRAYSCALE)
        _, mask = cv2.threshold(mask, 127, 255, cv2.THRESH_BINARY)
        
        if img.shape[:2] != mask.shape:
            mask = cv2.resize(mask, (img.shape[1], img.shape[0]), interpolation=cv2.INTER_NEAREST)
            
        img_combined = cv2.bitwise_and(img, img, mask=mask)
        img_combined = cv2.cvtColor(img_combined, cv2.COLOR_BGR2RGB)
        pil_img = Image.fromarray(img_combined)

        if self.transform: 
            img_tensor = self.transform(pil_img)

        msa_data = np.load(item['npy_path']).astype(np.float32)
        msa_data = msa_data / (msa_data.max() + 1e-8)
        msa_tensor = torch.from_numpy(msa_data).unsqueeze(0)

        return img_tensor, item['label'], msa_tensor

class HybridDistillationLoss(nn.Module):
    def __init__(self, alpha, beta):
        super(HybridDistillationLoss, self).__init__()
        self.alpha = alpha
        self.beta = beta
        self.mse = nn.MSELoss()
        self.l1 = nn.L1Loss()

    def forward(self, student_feat, teacher_feat):
        return self.alpha * self.mse(student_feat, teacher_feat) + self.beta * self.l1(student_feat, teacher_feat)

class FocalLoss(nn.Module):
    def __init__(self, weight=None, gamma=2.0):
        super(FocalLoss, self).__init__()
        self.weight = weight
        self.gamma = gamma

    def forward(self, inputs, targets):
        ce_loss = F.cross_entropy(inputs, targets, weight=self.weight, reduction='none')
        pt = torch.exp(-ce_loss)
        return ((1 - pt) ** self.gamma * ce_loss).mean()

class SADNet_EfficientNet(nn.Module):
    def __init__(self):
        super(SADNet_EfficientNet, self).__init__()
        effnet = models.efficientnet_v2_s(weights=models.EfficientNet_V2_S_Weights.DEFAULT)
        self.features = effnet.features
        
        num_ftrs = effnet.classifier[1].in_features 
        
        self.distill_branch = nn.Sequential(
            nn.Conv2d(num_ftrs, 256, kernel_size=3, padding=1),
            nn.BatchNorm2d(256),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 1, kernel_size=1) 
        )
        
        self.fc = nn.Sequential(nn.Dropout(0.2), nn.Linear(num_ftrs, 2))

    def forward(self, x):
        f = self.features(x)
        
        distill_out = self.distill_branch(f)
        distill_out = F.interpolate(distill_out, size=(256, 256), mode='bilinear', align_corners=False)
        
        pooled = F.adaptive_avg_pool2d(f, (1, 1)).view(x.size(0), -1)
        class_out = self.fc(pooled)
        
        return class_out, distill_out

def train_and_eval(model, train_loader, val_loader, alpha, beta, epochs, weight_path):
    cls_weights = torch.tensor([0.778, 1.397], dtype=torch.float32).to(device)
    criterion_cls = FocalLoss(weight=cls_weights, gamma=2.0)
    criterion_kd = HybridDistillationLoss(alpha, beta)
    
    optimizer = optim.AdamW(model.parameters(), lr=0.0001, weight_decay=0.03)
    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    
    best_acc = 0.0
    lambda_distill = 0.5 

    for epoch in range(epochs):
        model.train()
        for inputs, labels, msa_targets in train_loader:
            inputs, labels, msa_targets = inputs.to(device), labels.to(device), msa_targets.to(device)
            optimizer.zero_grad()
            
            class_preds, distill_preds = model(inputs)
            loss_cls = criterion_cls(class_preds, labels)
            loss_kd = criterion_kd(distill_preds, msa_targets)
            
            total_loss = loss_cls + lambda_distill * loss_kd
            total_loss.backward()
            optimizer.step()
        scheduler.step()

        model.eval()
        correct, total = 0, 0
        with torch.no_grad():
            for inputs, labels, _ in val_loader:
                inputs, labels = inputs.to(device), labels.to(device)
                outputs, _ = model(inputs)
                _, preds = torch.max(outputs, 1)
                total += labels.size(0)
                correct += torch.sum(preds == labels.data).item()
        
        val_acc = correct / total
        if val_acc > best_acc:
            best_acc = val_acc
            torch.save(model.state_dict(), weight_path)

    model.load_state_dict(torch.load(weight_path, map_location=device))
    model.eval()
    all_preds, all_labels_eval, all_probs = [], [], []
    total_samples, test_running_loss, total_inf_time = len(val_loader.dataset), 0.0, 0.0
    
    _ = model(torch.randn(1, 3, 256, 256).to(device))
    
    with torch.no_grad():
        for inputs, labels, msa_targets in val_loader:
            inputs, labels, msa_targets = inputs.to(device), labels.to(device), msa_targets.to(device)
            
            torch.cuda.synchronize()
            start_time = time.perf_counter()
            outputs, distill_preds = model(inputs)
            torch.cuda.synchronize()
            total_inf_time += (time.perf_counter() - start_time)
            
            loss = criterion_cls(outputs, labels) + lambda_distill * criterion_kd(distill_preds, msa_targets)
            test_running_loss += loss.item() * inputs.size(0)
            
            probs = torch.nn.functional.softmax(outputs, dim=1)[:, 1] 
            _, preds = torch.max(outputs, 1)
            
            all_preds.extend(preds.cpu().numpy())
            all_labels_eval.extend(labels.cpu().numpy())
            all_probs.extend(probs.cpu().numpy())
            
    cm = confusion_matrix(all_labels_eval, all_preds, labels=[0, 1])
    tn, fp, fn, tp = cm.ravel()
    
    acc = (tp + tn) / total_samples
    se = tp / (tp + fn) if (tp + fn) > 0 else 0.0
    sp = tn / (tn + fp) if (tn + fp) > 0 else 0.0
    auc = roc_auc_score(all_labels_eval, all_probs)
    f1 = f1_score(all_labels_eval, all_preds)
    time_ms = (total_inf_time / total_samples) * 1000
    
    return acc, se, sp, f1, auc, test_running_loss/total_samples, time_ms
