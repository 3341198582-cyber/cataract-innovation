import cv2
import os
import numpy as np

base_src = '/home/ubuntu/cataract_project_data'
target_size = 224

print("===Step 2: 重新执行标准化 ROI 裁剪 (224x224) ===\n")

for sp in ['train', 'test']:
    for cls in ['cataract', 'normal']:
        src_dir = os.path.join(base_src, sp, cls)
        pupil_dir = os.path.join(src_dir, f"{cls}_pupil")
        roi_dir = os.path.join(src_dir, f"{cls}_ROI")
        
        if not os.path.exists(pupil_dir): continue
        os.makedirs(roi_dir, exist_ok=True)
        os.system(f'chmod 777 "{roi_dir}"')

        files = sorted([f for f in os.listdir(pupil_dir) if f.lower().endswith(('.png', '.jpg'))])
        print(f"正在裁剪: {sp}/{cls} (共 {len(files)} 张)...")

        for f in files:
            img = cv2.imread(os.path.join(src_dir, f))
            mask = cv2.imread(os.path.join(pupil_dir, f), cv2.IMREAD_GRAYSCALE)
            
            if img is None or mask is None: continue
            h, w = img.shape[:2]

            # 读取 Step1 的精准掩膜中心
            y_coords, x_coords = np.where(mask > 127)
            if len(y_coords) > 0:
                cx, cy = int(np.mean(x_coords)), int(np.mean(y_coords))
            else:
                cx, cy = w // 2, h // 2

            side = min(h, w)
            half = side // 2

            x1, y1 = cx - half, cy - half
            x2, y2 = cx + half, cy + half

            # 边界推回
            if x1 < 0: x2 -= x1; x1 = 0
            if y1 < 0: y2 -= y1; y1 = 0
            if x2 > w: x1 -= (x2 - w); x2 = w
            if y2 > h: y1 -= (y2 - h); y2 = h

            cropped = img[y1:y2, x1:x2]
            if cropped.size > 0:
                img_224 = cv2.resize(cropped, (target_size, target_size), interpolation=cv2.INTER_AREA)
                # 自动覆盖旧文件
                cv2.imwrite(os.path.join(roi_dir, f), img_224)

print("\nStep 2ROI")
